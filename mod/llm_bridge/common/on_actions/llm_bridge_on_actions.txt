# LLM Bridge -- when telemetry fires.
#
# Daily is the default because the harness decides its own cadence and would
# rather throw away lines than miss a transition. If logging volume becomes a
# problem, move llm_bridge_emit_state to on_weekly and keep the event hooks
# daily: events are what wake the planner early, and they are cheap.

on_actions = {
	on_startup = {
		effect = {
			every_country = {
				limit = { is_ai = no }
				set_variable = { llmb_ai_delegated_armies = 0 }
				log = "LLMB|v3|boot|tag=[THIS.GetTag]|start=[GetDateText]"
				llm_bridge_emit_state = yes
			}
		}
	}

	# on_daily fires once per country, spread over the day's hours, with ROOT
	# set to that country. Emit only on the player's own tick; looping
	# every_country here logged the player's numbers once per country per day,
	# under the ticking country's tag.
	on_daily = {
		effect = {
			if = {
				limit = { is_ai = no }
				llm_bridge_emit_state = yes
			}
		}
	}

	# --- events that should cut a scheduled wait short -----------------------

	# Both fire for every war and surrender in the world, in the declarer's (or
	# the capitulating country's) scope, with FROM the other side. Seen live on
	# 1.19.5: AI-vs-AI wars filled the log. Only the player's own get a line.
	on_declare_war = {
		effect = {
			if = {
				limit = { OR = { is_ai = no FROM = { is_ai = no } } }
				set_variable = { llmb_event_kind = 1 }   # war_declared
				llm_bridge_emit_war = yes
			}
		}
	}

	on_capitulation = {
		effect = {
			if = {
				limit = { OR = { is_ai = no FROM = { is_ai = no } } }
				set_variable = { llmb_event_kind = 2 }   # capitulation
				llm_bridge_emit_event = yes
			}
		}
	}

	# Focus, research and construction completion have no on_action in 1.19.5
	# (only the military-industrial-organisation ones do). The harness derives
	# them from the daily state line and the probe lines instead: see
	# scripted_effects/llm_bridge_probe.txt and docs/mod-bridge.md.
}
