# LLM Bridge -- the front picture: divisions per state (#85)
#
# A division scope has no location a script can read. documentation/triggers_
# documentation.md lists for it only unit_strength (l.7773), unit_organization
# (l.7766), has_officer_name (l.4542), division_has_battalion_in_template
# (l.2647) and division_has_majority_template (l.2656); effects_documentation.md
# only set_unit_organization, destroy_unit and a few more (see docs/unit-orders.md).
# What a script CAN do is ask a state for the divisions standing in it:
# every_state_division (effects l.4174, STATE scope), the way the game's own
# 00_scripted_effects.txt does for paratroopers. So the front picture is per state:
#
#   LLMB|v5|div|tag=SWE|state=64|ctl=GER|own=3|own_weak=1|foe=5|foe_weak=0
#   LLMB|v5|divend|tag=SWE|n=7
#
# One div line per state that holds a division of either side, then one divend
# line with how many div lines this run printed (the reader keeps the last n and
# drops a run it saw only part of). "foe" is a division whose owner is at war with
# the player; "weak" is unit_strength < 0.5. Allied divisions are not counted.
#
# Which states: those the enemy controls, and the player's own states that touch
# an enemy-controlled one. That is the front and the ground behind it, bounded by
# the war, and costs nothing in peace (the has_war guard).
#
# VERIFY IN A LIVE GAME (docs/live-session.md): ROOT inside every_state is the
# player; OWNER works in a division scope (the game's own script uses it there);
# [THIS.GetID] and [THIS.Controller.GetTag] expand in a state scope (loc_objects_
# documentation.md l.629, l.645). A token the game does not expand prints
# literally, which the reader treats as a bad line and drops.
#
# Called from on_daily for the player (llm_bridge_front_on_actions.txt) and on
# demand: `effect <TAG> llm_bridge_emit_front`.

llm_bridge_emit_front = {
	set_temp_variable = { llmb_front_n = 0 }
	if = {
		limit = { has_war = yes }
		every_state = {
			limit = {
				OR = {
					CONTROLLER = { has_war_with = ROOT }
					AND = {
						is_controlled_by = ROOT
						any_neighbor_state = { CONTROLLER = { has_war_with = ROOT } }
					}
				}
			}
			set_temp_variable = { llmb_own = 0 }
			set_temp_variable = { llmb_own_weak = 0 }
			set_temp_variable = { llmb_foe = 0 }
			set_temp_variable = { llmb_foe_weak = 0 }
			every_state_division = {
				limit = { OWNER = { tag = ROOT } }
				add_to_temp_variable = { llmb_own = 1 }
				if = {
					limit = { unit_strength < 0.5 }
					add_to_temp_variable = { llmb_own_weak = 1 }
				}
			}
			every_state_division = {
				limit = { OWNER = { has_war_with = ROOT } }
				add_to_temp_variable = { llmb_foe = 1 }
				if = {
					limit = { unit_strength < 0.5 }
					add_to_temp_variable = { llmb_foe_weak = 1 }
				}
			}
			if = {
				limit = {
					OR = {
						check_variable = { llmb_own > 0 }
						check_variable = { llmb_foe > 0 }
					}
				}
				# log reads variables of the scope it runs in, so copy the counts
				# into this state's own variables first.
				set_variable = { llmb_f_own = llmb_own }
				set_variable = { llmb_f_own_weak = llmb_own_weak }
				set_variable = { llmb_f_foe = llmb_foe }
				set_variable = { llmb_f_foe_weak = llmb_foe_weak }
				add_to_temp_variable = { llmb_front_n = 1 }
				log = "LLMB|v5|div|tag=[ROOT.GetTag]|state=[THIS.GetID]|ctl=[THIS.Controller.GetTag]|own=[?llmb_f_own]|own_weak=[?llmb_f_own_weak]|foe=[?llmb_f_foe]|foe_weak=[?llmb_f_foe_weak]"
			}
		}
	}
	set_variable = { llmb_front_states = llmb_front_n }
	log = "LLMB|v5|divend|tag=[THIS.GetTag]|n=[?llmb_front_states]"
}
